Update ShouldAutoCreateShipment to also consider fulfillment type. The cleanest way is to change its signature to take the whole payload, then check both status and fulfillment:
csharp        private bool ShouldAutoCreateShipment(PluginWebhookPayload payload)
        {
            // Don't book a courier shipment for orders the customer is
            // collecting in person — there's no delivery to schedule.
            // 'special_trip' is also collection-flow (no scheduled route),
            // but is intentionally still handled by Shiplogic as an ad-hoc
            // booking, so we treat it as bookable.
            if (string.Equals(payload.FulfillmentType, "collect", StringComparison.OrdinalIgnoreCase))
            {
                _logger.LogInformation(
                    "Skipping Shiplogic shipment creation for collect order. WooOrderID: {WooOrderId}",
                    payload.WooOrderId);
                return false;
            }

            var paidStatuses = new[] { "processing", "completed" };
            return paidStatuses.Contains(payload.Status?.ToLower());
        }
